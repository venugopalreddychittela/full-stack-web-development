"Write MongoDB queries to perform the following operations on a students collection: (a) Insert a student record, (b) retrieve all student records, (c) update a student's department, and (d) delete a student record."

A university department maintains student academic records, branch assignments, semester enrollment, and GPA metrics. Previously handled using manual registers, the administration requires a modern, web-based Student Management System accessible globally over the internet.

The application must satisfy the complete 20-mark assessment rubric through the following requirements:

  • Store student details including Roll Number (rollNo), Student Name (name), Academic Department (department), Current Semester (semester), Cumulative Grade Point Average (gpa), and Institutional Email (email).

Field Design Specification:
Field Suggested Data Type Purpose & Constraints
rollNo String Unique student registration identifier (e.g., "A24126552078"). Unique index applied.
name String Full legal name of the student. Required, trimmed string.
department String Academic branch (e.g., Computer Science, Information Technology). Indexed.
semester Number (Integer) Current semester of study (Valid range: 1 to 8).
gpa Number (Double) Cumulative Grade Point Average (Valid range: 0.00 to 10.00).
email String Official university communication email address.
_id ObjectId System-generated 12-byte BSON primary key automatically created by MongoDB.

backend/server.js:

require("dotenv").config();
const express = require("express");
const mongoose = require("mongoose");
const cors = require("cors");
const app = express();
app.use(express.json());

const allowedOrigins = (process.env.FRONTEND_ORIGIN || "http://localhost:5173").split(",").map(o => o.trim());
app.use(cors({ origin: allowedOrigins }));

const studentSchema = new mongoose.Schema({
  rollNo: { type: String, required: true, unique: true, trim: true },
  name: { type: String, required: true, trim: true },
  department: { type: String, required: true, trim: true },
  semester: { type: Number, required: true, min: 1, max: 8 },
  gpa: { type: Number, required: true, min: 0.0, max: 10.0 },
  email: { type: String, required: true, trim: true }
}, { timestamps: true });
studentSchema.index({ department: 1 });
const Student = mongoose.model("Student", studentSchema);

app.get("/api/health", (req, res) => res.json({ status: "ok", message: "Student Management System API is running" }));

// Q4 (b) Retrieve all students
app.get("/api/students", async (req, res) => {
  try { res.json(await Student.find().sort({ rollNo: 1 })); }
  catch (err) { res.status(500).json({ message: "Unable to retrieve student records" }); }
});

// Q4 (a) Insert a student record
app.post("/api/students", async (req, res) => {
  try { res.status(201).json(await Student.create(req.body)); }
  catch (err) { res.status(400).json({ message: err.message }); }
});

// Q4 (c) Update student details
app.put("/api/students/:id", async (req, res) => {
  try {
    const s = await Student.findByIdAndUpdate(req.params.id, req.body, { new: true, runValidators: true });
    if (!s) return res.status(404).json({ message: "Student not found" });
    res.json(s);
  } catch (err) { res.status(400).json({ message: err.message }); }
});

// Q4 (c) Dedicated endpoint to update student's department by Roll Number
app.patch("/api/students/:rollNo/department", async (req, res) => {
  try {
    const s = await Student.findOneAndUpdate({ rollNo: req.params.rollNo }, { $set: { department: req.body.department } }, { new: true });
    if (!s) return res.status(404).json({ message: "Student not found" });
    res.json({ message: "Department updated successfully", student: s });
  } catch (err) { res.status(400).json({ message: err.message }); }
});

// Q4 (d) Delete a student record
app.delete("/api/students/:id", async (req, res) => {
  try {
    const s = await Student.findByIdAndDelete(req.params.id);
    if (!s) return res.status(404).json({ message: "Student not found" });
    res.json({ message: "Student record deleted successfully" });
  } catch (err) { res.status(400).json({ message: err.message }); }
});

// Aggregation summary route
app.get("/api/reports/department-summary", async (req, res) => {
  try {
    const summary = await Student.aggregate([
      { $group: { _id: "$department", studentCount: { $sum: 1 }, averageGpa: { $avg: "$gpa" }, highestGpa: { $max: "$gpa" }, lowestGpa: { $min: "$gpa" } } },
      { $sort: { studentCount: -1 } }
    ]);
    res.json(summary);
  } catch (err) { res.status(500).json({ message: "Unable to generate summary" }); }
});

const PORT = process.env.PORT || 5000;
async function start() {
  try {
    await mongoose.connect(process.env.MONGODB_URI, { dbName: process.env.DB_NAME || "student_mgmt_db" });
    app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
  } catch (err) { console.error("Database connection failed:", err.message); process.exit(1); }
}
start();

8. Develop the React Frontend

Create Vite React project: npm create vite@latest frontend -- --template react && cd frontend && npm install
Set frontend/.env: VITE_API_URL=http://localhost:5000

frontend/src/App.jsx:

import { useEffect, useState } from "react";
import "./style.css";

const API = (import.meta.env.VITE_API_URL || "http://localhost:5000").replace(/\/$/, "") + "/api";
const emptyForm = { rollNo: "", name: "", department: "Computer Science and Engineering", semester: 5, gpa: 8.5, email: "" };

export default function App() {
  const [students, setStudents] = useState([]);
  const [summary, setSummary] = useState([]);
  const [form, setForm] = useState(emptyForm);
  const [editingId, setEditingId] = useState(null);
  const [message, setMessage] = useState("");
  const [quickRoll, setQuickRoll] = useState("");
  const [quickDept, setQuickDept] = useState("Artificial Intelligence & Data Science");

  async function loadData() {
    try {
      const [sRes, sumRes] = await Promise.all([fetch(`${API}/students`), fetch(`${API}/reports/department-summary`)]);
      if (sRes.ok) setStudents(await sRes.json());
      if (sumRes.ok) setSummary(await sumRes.json());
    } catch (err) { setMessage("API Connection Error"); }
  }

  useEffect(() => { loadData(); }, []);

  function handleChange(e) {
    const { name, value } = e.target;
    setForm({ ...form, [name]: name === "semester" || name === "gpa" ? Number(value) : value });
  }

  async function handleSubmit(e) {
    e.preventDefault();
    const url = editingId ? `${API}/students/${editingId}` : `${API}/students`;
    const res = await fetch(url, { method: editingId ? "PUT" : "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(form) });
    if (!res.ok) { setMessage("Save failed"); return; }
    setMessage(editingId ? "Student updated" : "Student added");
    setForm(emptyForm); setEditingId(null); await loadData();
  }

  async function updateDepartment(e) {
    e.preventDefault();
    const res = await fetch(`${API}/students/${quickRoll}/department`, { method: "PATCH", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ department: quickDept }) });
    if (!res.ok) { setMessage("Department update failed"); return; }
    setMessage(`Updated department for ${quickRoll}`); setQuickRoll(""); await loadData();
  }

  async function removeStudent(id) {
    if (!window.confirm("Delete record?")) return;
    const res = await fetch(`${API}/students/${id}`, { method: "DELETE" });
    if (!res.ok) { setMessage("Delete failed"); return; }
    setMessage("Student deleted"); await loadData();
  }

  return (
    <main className="container">
      <h1>Student Management System</h1>
      <p>Candidate Roll No: A24126552078 | Question 4: MongoDB Operations</p>
      {message && <p className="alert">{message}</p>}

      {/* Q4 (a) Insert Form */}
      <form onSubmit={handleSubmit}>
        <input name="rollNo" value={form.rollNo} onChange={handleChange} placeholder="Roll No" required />
        <input name="name" value={form.name} onChange={handleChange} placeholder="Name" required />
        <input name="department" value={form.department} onChange={handleChange} placeholder="Department" required />
        <input type="number" name="semester" value={form.semester} onChange={handleChange} />
        <input type="number" step="0.01" name="gpa" value={form.gpa} onChange={handleChange} />
        <input type="email" name="email" value={form.email} onChange={handleChange} placeholder="Email" />
        <button type="submit">{editingId ? "Update" : "Add Student"}</button>
      </form>

      {/* Q4 (c) Quick Department Update Form */}
      <form onSubmit={updateDepartment}>
        <input value={quickRoll} onChange={(e) => setQuickRoll(e.target.value)} placeholder="Roll No to Update" required />
        <input value={quickDept} onChange={(e) => setQuickDept(e.target.value)} placeholder="New Department" required />
        <button type="submit">Update Department (Q4-c)</button>
      </form>

      {/* Q4 (b) Retrieve & (d) Delete Table */}
      <table>
        <thead><tr><th>Roll No</th><th>Name</th><th>Department</th><th>Sem</th><th>GPA</th><th>Actions</th></tr></thead>
        <tbody>
          {students.map((s) => (
            <tr key={s._id}>
              <td>{s.rollNo}</td><td>{s.name}</td><td>{s.department}</td><td>{s.semester}</td><td>{s.gpa}</td>
              <td><button onClick={() => { setEditingId(s._id); setForm(s); }}>Edit</button>
                  <button onClick={() => removeStudent(s._id)}>Delete</button></td>
            </tr>
          ))}
        </tbody>
      </table>
    </main>
  );
}
